document.getElementById("telegram-button").addEventListener("click", async function () { // Generate Telegram message using your existing function const message = getTelegramMessage(); // Build the PDF content (including extended details) buildPdfContent(); const pdfPreview = document.getElementById("pdf-preview"); if (!pdfPreview) { alert("PDF preview element not found."); return; } // Temporarily reveal the PDF container so html2pdf can render it pdfPreview.classList.remove("hidden-pdf"); pdfPreview.style.visibility = "visible"; const opt = { margin: 10, filename: 'CharacterDetails.pdf', image: { type: 'jpeg', quality: 0.98 }, html2canvas: { scale: 2, windowWidth: document.documentElement.offsetWidth }, jsPDF: { unit: 'mm', format: 'a4', orientation: 'portrait' } }; try { // Instead of automatically triggering download, output as a Blob await html2pdf().set(opt).from(pdfPreview).outputPdf('blob').then(function(pdfBlob) { const blobUrl = URL.createObjectURL(pdfBlob); // Mobile fallback: If on iOS, display a link for long press to download if (isIOS()) { // Remove any existing fallback link first let existingLink = document.getElementById("mobile-download-link"); if (existingLink) { existingLink.remove(); } const mobileLink = document.createElement("a"); mobileLink.href = blobUrl; mobileLink.id = "mobile-download-link"; mobileLink.textContent = "Long press here to save your PDF"; mobileLink.style.display = "block"; mobileLink.style.padding = "10px"; mobileLink.style.backgroundColor = "#eee"; mobileLink.style.marginTop = "10px"; // Append the link inside the PDF container so the user sees it pdfPreview.appendChild(mobileLink); } else { // For non-mobile devices, trigger download automatically const a = document.createElement("a"); a.href = blobUrl; a.download = 'CharacterDetails.pdf'; document.body.appendChild(a); a.click(); document.body.removeChild(a); } }); alert("PDF generated successfully!"); } catch (error) { console.error("PDF generation failed:", error); alert("PDF download failed! Please ensure the content is loaded."); } // Restore the hidden state of the PDF container (the fallback link remains visible on mobile) pdfPreview.classList.add("hidden-pdf"); // For iOS, delay sending the Telegram message so the PDF generation doesn't interfere if (isIOS()) { setTimeout(async () => { await sendTelegramMessage(message); }, 1000); } else { await sendTelegramMessage(message); } }); // Helper to detect iOS function isIOS() { return /iPad|iPhone|iPod/.test(navigator.userAgent) && !window.MSStream; }
